Credit Card Fraud — EDA & ML Pipeline

End-to-end exploration and modelling of the Kaggle Credit Card Fraud dataset (284K transactions, 0.17% fraud rate). Class-imbalance handling, XGBoost + isolation forest ensemble, and SHAP feature attribution.

Project Goal

The Kaggle Credit Card Fraud dataset contains transactions made by European cardholders in September 2013. Only 492 of 284,807 transactions are fraudulent (0.17%) — making this a severe class imbalance problem.

The goal was to build a rigorous ML pipeline: thorough EDA on PCA-transformed features, multiple imbalance-handling strategies, hyperparameter optimisation, and SHAP-based explainability to understand which signals actually drive fraud detection.

Pipeline Overview

EDA + Distribution Analysis → SMOTE + Class Weights → Optuna XGBoost Tuning → Isolation Forest Baseline → Ensemble Scoring → SHAP Explainability

01_eda.py — Class imbalance analysis and feature correlations Python · Pandas · Matplotlib · Seaborn 
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

df = pd.read_csv("creditcard.csv")

# ── Imbalance overview ──
print(df['Class'].value_counts(normalize=True).round(4))
# 0 (legit)  → 99.83%
# 1 (fraud)  →  0.17%

# ── Amount distribution: fraud vs. legit ──
fig, axes = plt.subplots(1, 2, figsize=(14, 4))
for cls, label, ax in zip([0, 1], ['Legit', 'Fraud'], axes):
    df[df.Class == cls]['Amount'].hist(bins=50, ax=ax, log=True)
    ax.set_title(f'{label} — Transaction Amount Distribution')
    ax.set_xlabel('Amount (EUR)')
plt.tight_layout()
plt.savefig("amount_dist.png", dpi=150)

# ── Time pattern: fraud concentrates in off-peak hours ──
df['hour'] = (df['Time'] % 86400) / 3600
fraud_hours = df[df.Class == 1]['hour']
print(f"Fraud hour mean: {fraud_hours.mean():.1f}h  "
      f"std: {fraud_hours.std():.1f}h")

# ── Feature correlation with target ──
corr = df.drop(columns=['Time']).corrwith(df['Class']).abs().sort_values(ascending=False)
print("Top 10 features correlated with fraud:")
print(corr.head(10))
02_model.py — SMOTE + XGBoost with Optuna hyperparameter search Python · XGBoost · Optuna · imbalanced-learn 
import xgboost as xgb
import optuna
from imblearn.over_sampling import SMOTE
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import average_precision_score
import numpy as np

# ── SMOTE resampling ──
smote = SMOTE(sampling_strategy=0.1, random_state=42, k_neighbors=5)
X_res, y_res = smote.fit_resample(X_train, y_train)
print(f"After SMOTE: {np.bincount(y_res)}")

# ── Optuna objective ──
def objective(trial):
    params = {
        "n_estimators":       trial.suggest_int("n_estimators", 200, 800),
        "max_depth":          trial.suggest_int("max_depth", 3, 9),
        "learning_rate":      trial.suggest_float("learning_rate", 0.01, 0.2, log=True),
        "subsample":          trial.suggest_float("subsample", 0.6, 1.0),
        "colsample_bytree":   trial.suggest_float("colsample_bytree", 0.6, 1.0),
        "scale_pos_weight":   trial.suggest_float("scale_pos_weight", 50, 200),
        "eval_metric":        "aucpr",
        "tree_method":        "hist",
        "random_state":       42,
    }
    cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    scores = []
    for train_idx, val_idx in cv.split(X_res, y_res):
        model = xgb.XGBClassifier(**params)
        model.fit(
            X_res[train_idx], y_res[train_idx],
            eval_set=[(X_res[val_idx], y_res[val_idx])],
            verbose=False,
        )
        preds = model.predict_proba(X_res[val_idx])[:, 1]
        scores.append(average_precision_score(y_res[val_idx], preds))
    return np.mean(scores)

study = optuna.create_study(direction="maximize")
study.optimize(objective, n_trials=60, show_progress_bar=True)
print(f"Best AUC-PR: {study.best_value:.4f}")
print("Best params:", study.best_params)
