Problem Statement

A FinTech client processing insurance and trade-finance claims had historically relied on structured tabular models. But fraudulent claims often left signals across multiple data modalities simultaneously: altered document images, inconsistent text narratives, and anomalous structured features — each weak alone but decisive when combined.

No single-modality model achieved acceptable precision at required recall thresholds. The goal was to build a system that could jointly reason over images, text, and tabular data in a single forward pass, enabling investigators to act on a single calibrated risk score.

System Architecture

Cheque / Doc Image → ResNet-50 (256-d)
Transaction Text → BERT [CLS] (256-d)
Structured Features → LightGBM Leaf Emb (256-d)
Stack (B, 3, 256) → Cross-Modal Attention (8 heads) → MLP Classifier → Risk Score

Each modality produces a 256-dimensional embedding. The three embeddings are stacked into a (B, 3, 256) tensor and passed through an 8-head cross-modal attention layer — letting modalities attend to each other before a final MLP makes the classification decision. This outperformed both concatenation and simple averaging fusion.

Key Implementation

branches.py — Vision branch (ResNet-50) and Text branch (BERT) Python · PyTorch · Hugging Face Transformers 
import torch.nn as nn
import torchvision.models as models
from transformers import BertModel

class VisionBranch(nn.Module):
    def __init__(self, out_dim=256, freeze_backbone=False):
        super().__init__()
        backbone     = models.resnet50(weights=models.ResNet50_Weights.IMAGENET1K_V2)
        self.feats   = nn.Sequential(*list(backbone.children())[:-1])  # drop FC
        if freeze_backbone:
            for p in self.feats.parameters(): p.requires_grad = False
        self.proj = nn.Sequential(
            nn.Flatten(), nn.Linear(2048, 512), nn.ReLU(),
            nn.Dropout(0.3), nn.Linear(512, out_dim),
        )
    def forward(self, img):
        return self.proj(self.feats(img))   # (B, out_dim)


class TextBranch(nn.Module):
    def __init__(self, out_dim=256, model_name="bert-base-uncased"):
        super().__init__()
        self.bert = BertModel.from_pretrained(model_name)
        self.proj = nn.Sequential(
            nn.Linear(768, 512), nn.ReLU(),
            nn.Dropout(0.2), nn.Linear(512, out_dim),
        )
    def forward(self, input_ids, attention_mask):
        cls = self.bert(input_ids, attention_mask).last_hidden_state[:, 0]
        return self.proj(cls)               # (B, out_dim)
fusion_model.py — Late-fusion with cross-modal attention Python · PyTorch 
import torch
import torch.nn as nn

class MultiModalFraudDetector(nn.Module):
    def __init__(self, tab_dim: int, fusion_dim: int = 256):
        super().__init__()
        self.vision = VisionBranch(out_dim=256)
        self.text   = TextBranch(out_dim=256)
        self.tab_proj = nn.Sequential(
            nn.Linear(tab_dim, 256), nn.ReLU(), nn.Dropout(0.3)
        )
        # Cross-modal attention: let modalities attend to each other
        self.cross_attn = nn.MultiheadAttention(
            256, num_heads=8, batch_first=True, dropout=0.1
        )
        self.classifier = nn.Sequential(
            nn.Linear(256 * 3, fusion_dim),
            nn.GELU(), nn.Dropout(0.3),
            nn.Linear(fusion_dim, 1),
        )

    def forward(self, img, input_ids, attention_mask, tab_feat):
        v = self.vision(img)                           # (B, 256)
        t = self.text(input_ids, attention_mask)       # (B, 256)
        x = self.tab_proj(tab_feat)                    # (B, 256)

        # Stack → cross-attend → flatten
        modal  = torch.stack([v, t, x], dim=1)         # (B, 3, 256)
        fused, _ = self.cross_attn(modal, modal, modal)
        flat   = fused.reshape(fused.shape[0], -1)     # (B, 768)
        return self.classifier(flat).squeeze(-1)
tabular_emb.py — LightGBM leaf-index encoding as tabular embedding Python · LightGBM 
import lightgbm as lgb
import numpy as np

def train_lgb_for_embedding(X_tab, y_tab):
    """Train LightGBM; use leaf indices as a rich embedding."""
    dtrain = lgb.Dataset(X_tab, label=y_tab)
    params = {
        "objective": "binary", "metric": "average_precision",
        "num_leaves": 127, "learning_rate": 0.05, "verbose": -1,
        "scale_pos_weight": (y_tab==0).sum() / (y_tab==1).sum(),
    }
    model = lgb.train(
        params, dtrain, num_boost_round=300,
        valid_sets=[dtrain],
        callbacks=[lgb.early_stopping(20), lgb.log_evaluation(50)],
    )
    return model

def get_leaf_embedding(lgb_model, X_tab) -> np.ndarray:
    """
    Each tree assigns a sample to one leaf → one-hot per tree.
    Concatenated leaf indices form a high-cardinality embedding
    that captures complex non-linear feature interactions.
    """
    return lgb_model.predict(X_tab, pred_leaf=True).astype(np.float32)
