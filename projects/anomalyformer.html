Problem Statement

An energy-sector client operated turbines and compressors instrumented with 32 sensor channels (vibration, temperature, pressure, RPM). Traditional threshold alarms only triggered at the moment of failure — by which point damage was already done. They needed early-warning detection from multivariate sensor patterns, but had no labelled anomaly data for supervised training.

The solution: train a reconstruction-based Transformer on normal operating behaviour. When operating conditions deviate from learned normality, reconstruction error spikes — providing an anomaly signal hours before physical failure.

System Architecture

32-channel Sensor Stream → Standard Scaler → Sliding Window (64 steps) → Patch Embedding (patch=8) → 4-Layer Transformer Encoder → Reconstruction Head → MSE Anomaly Score

The model uses a Patch-TST architecture: each 64-step window is split into 8-step patches, linearly projected to a 128-dimensional embedding, and fed into a 4-layer Transformer Encoder with pre-norm residuals. The decoder reconstructs the original window from the encoded patches.

During training (normal data only), the model learns to reconstruct healthy patterns. At inference, per-timestep MSE averaged across sensor channels becomes the anomaly score. A threshold at the 99th percentile of training reconstruction errors triggers an alert, providing leads to the maintenance team with average 4-hour advance warning.

Key Implementation

dataset.py — Sliding-window sensor dataset with StandardScaler Python · PyTorch 
import numpy as np
import torch
from torch.utils.data import Dataset
from sklearn.preprocessing import StandardScaler

class SensorDataset(Dataset):
    def __init__(self, data: np.ndarray, seq_len: int = 64, stride: int = 8):
        """
        data : shape (T, N_sensors)
        Fit scaler on this split; call .transform() for test data.
        """
        self.seq_len = seq_len
        self.scaler  = StandardScaler()
        scaled       = self.scaler.fit_transform(data)

        # Sliding window patches
        windows = []
        for i in range(0, len(scaled) - seq_len, stride):
            windows.append(scaled[i : i + seq_len])
        self.windows = np.array(windows, dtype=np.float32)

    def __len__(self):         return len(self.windows)
    def __getitem__(self, i):  return torch.tensor(self.windows[i])
model.py — Patch-TST Transformer with reconstruction head Python · PyTorch 
import torch
import torch.nn as nn

class AnomalyFormer(nn.Module):
    def __init__(self, n_sensors=32, seq_len=64, patch_len=8,
                 d_model=128, n_heads=8, n_layers=4, dropout=0.1):
        super().__init__()
        self.patch_len = patch_len
        n_patches      = seq_len // patch_len

        # Patch projection: (B, n_patches, patch_len * N) → (B, n_patches, d_model)
        self.patch_proj = nn.Linear(patch_len * n_sensors, d_model)

        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=n_heads,
            dim_feedforward=d_model * 4,
            dropout=dropout, batch_first=True, norm_first=True,
        )
        self.encoder = nn.TransformerEncoder(encoder_layer, num_layers=n_layers)

        # Reconstruction: (B, n_patches, d_model) → (B, seq_len, n_sensors)
        self.reconstructor = nn.Linear(d_model, patch_len * n_sensors)
        self.n_patches  = n_patches
        self.n_sensors  = n_sensors

    def forward(self, x):
        # x: (B, T, N)
        B, T, N = x.shape
        # Patchify
        p = x.unfold(1, self.patch_len, self.patch_len)   # (B, n_p, N, p)
        p = p.permute(0, 1, 3, 2).reshape(B, self.n_patches, -1)
        # Encode
        emb   = self.patch_proj(p)                         # (B, n_p, d_model)
        enc   = self.encoder(emb)                          # (B, n_p, d_model)
        # Reconstruct
        recon = self.reconstructor(enc)                    # (B, n_p, p*N)
        recon = recon.view(B, self.n_patches, self.patch_len, N)
        recon = recon.reshape(B, T, N)
        return recon
detect.py — Threshold-based anomaly scoring from reconstruction error Python · PyTorch · NumPy 
import torch.nn.functional as F
import numpy as np

def fit_threshold(model, train_loader, percentile=99):
    """Compute anomaly threshold from normal training data."""
    model.eval()
    errors = []
    with torch.no_grad():
        for batch in train_loader:
            recon = model(batch)
            mse   = F.mse_loss(recon, batch, reduction='none')
            errors.append(mse.mean(dim=-1).cpu().numpy())  # (B, T)
    errors    = np.concatenate(errors)
    threshold = np.percentile(errors, percentile)
    print(f"Anomaly threshold (p{percentile}): {threshold:.6f}")
    return threshold

def score_stream(model, test_loader, threshold):
    model.eval()
    scores, flags = [], []
    with torch.no_grad():
        for batch in test_loader:
            recon = model(batch)
            mse   = F.mse_loss(recon, batch, reduction='none').mean(-1)
            scores.append(mse.cpu().numpy())
            flags.append((mse.cpu().numpy() > threshold))
    return np.concatenate(scores), np.concatenate(flags)
