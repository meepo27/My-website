Gradient-boosted stacking ensemble on the Home Credit Default Risk dataset (307K customers, 122 features). Advanced feature engineering from bureau and instalment tables; LightGBM + XGBoost base learners with logistic regression meta-learner.

Project Goal

The Home Credit Default Risk competition challenges participants to predict whether a loan applicant will repay their loan, using a rich set of application, bureau, credit card, and instalment history tables.

Beyond a single model, the goal was to build a full feature engineering pipeline that aggregates information across all related tables, a two-level stacking ensemble, and calibrated output probabilities suitable for credit risk band assignment.

feature_engineering.py — Aggregating bureau and instalment tables Python · Pandas · NumPy 
import pandas as pd
import numpy as np

def aggregate_bureau(bureau: pd.DataFrame, bureau_bal: pd.DataFrame) -> pd.DataFrame:
    """Aggregate credit bureau data per applicant."""
    # Join bureau balance status counts
    bal_counts = (bureau_bal
                  .groupby('SK_ID_BUREAU')['STATUS']
                  .value_counts()
                  .unstack(fill_value=0)
                  .add_prefix('bal_status_'))
    bureau = bureau.join(bal_counts, on='SK_ID_BUREAU')

    agg = {
        'DAYS_CREDIT':          ['min', 'max', 'mean'],
        'CREDIT_DAY_OVERDUE':   ['max', 'mean'],
        'AMT_CREDIT_SUM':       ['sum', 'mean'],
        'AMT_CREDIT_SUM_DEBT':  ['sum', 'mean'],
        'CNT_CREDIT_PROLONG':   'sum',
        # Bad debt ratio per applicant
        'AMT_CREDIT_SUM_DEBT':  'sum',
    }
    bureau_feats = (bureau
                    .groupby('SK_ID_CURR')
                    .agg(agg)
                    .reset_index())
    bureau_feats.columns = ['SK_ID_CURR'] + [
        f"bureau_{col[0]}_{col[1]}" for col in bureau_feats.columns[1:]
    ]

    # Bad debt ratio
    bureau_feats['bureau_bad_debt_ratio'] = (
        bureau_feats['bureau_AMT_CREDIT_SUM_DEBT_sum'] /
        (bureau_feats['bureau_AMT_CREDIT_SUM_sum'] + 1e-8)
    )
    return bureau_feats


def aggregate_installments(inst: pd.DataFrame) -> pd.DataFrame:
    """Payment behaviour features from instalment history."""
    inst['PAYMENT_DIFF']  = inst['AMT_INSTALMENT'] - inst['AMT_PAYMENT']
    inst['DAYS_LATE']     = (inst['DAYS_ENTRY_PAYMENT'] - inst['DAYS_INSTALMENT']).clip(lower=0)
    inst['IS_LATE']       = (inst['DAYS_LATE'] > 0).astype(int)

    return inst.groupby('SK_ID_CURR').agg(
        inst_payment_diff_mean  = ('PAYMENT_DIFF', 'mean'),
        inst_payment_diff_max   = ('PAYMENT_DIFF', 'max'),
        inst_days_late_mean     = ('DAYS_LATE', 'mean'),
        inst_late_rate          = ('IS_LATE', 'mean'),
        inst_count              = ('AMT_INSTALMENT', 'count'),
    ).reset_index()
stacking.py — Two-level LightGBM + XGBoost → logistic regression ensemble Python · LightGBM · XGBoost · Scikit-learn 
import lightgbm as lgb
import xgboost as xgb
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import roc_auc_score
import numpy as np

def get_oof_predictions(model_fn, X, y, n_splits=5):
    """Generate out-of-fold predictions for stacking."""
    oof = np.zeros(len(X))
    cv  = StratifiedKFold(n_splits=n_splits, shuffle=True, random_state=42)

    for fold, (tr_idx, val_idx) in enumerate(cv.split(X, y)):
        model = model_fn()
        model.fit(
            X[tr_idx], y[tr_idx],
            eval_set=[(X[val_idx], y[val_idx])],
        )
        oof[val_idx] = model.predict_proba(X[val_idx])[:, 1]
        print(f"  Fold {fold+1} AUC: {roc_auc_score(y[val_idx], oof[val_idx]):.4f}")

    print(f"  OOF AUC: {roc_auc_score(y, oof):.4f}")
    return oof

# Level-1: LightGBM
lgb_oof = get_oof_predictions(
    lambda: lgb.LGBMClassifier(
        n_estimators=2000, learning_rate=0.05, num_leaves=63,
        scale_pos_weight=11, n_jobs=-1,
